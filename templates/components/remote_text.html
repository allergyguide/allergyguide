{% component remote_text(src, start=1, end=0, page="") %} {#- load_data uses different arguments based on whether it's a remote or local file -#} {%- if src is starting_with(pat="http") -%} {%- set response = load_data(url=src, format="plain", required=false) -%} {%- else -%} {#- Try to load the file from a relative path -#} {%- if page and page.colocated_path %} {%- set colocated_path = page.colocated_path -%} {%- else %} {%- set colocated_path = "" -%} {%- endif %} {%- set relative_path = colocated_path ~ src -%} {%- set response = load_data(path=relative_path, format="plain", required=false) -%} {#- If relative path fails, try absolute path -#} {%- if not response -%} {%- set response = load_data(path=src, format="plain") -%} {%- endif -%} {%- endif -%} {%- if response -%} {%- set lines = response | trim_end | split(pat="\n") -%} {%- if start > 0 -%} {%- set start = start - 1 -%} {%- endif -%} {%- if end == 0 or end > lines | length -%} {%- set end = lines | length -%} {%- endif -%} {%- set lines = lines[start:end] -%} {{- lines | join(sep="\n") | safe -}} {%- else -%} {%- endif -%} {% endcomponent remote_text %}
